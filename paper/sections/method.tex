\section{Method}
\label{sec:method}

\subsection{Overview}
\label{sec:method:overview}
Algorithm~\ref{alg:planner} summarises the planner. An offline stage
decomposes $\CSfree$ into convex polytopes (Section~\ref{sec:method:decomp}),
constructs polytopic grasp and placement spaces
(Section~\ref{sec:method:grasp}), and intersects them to obtain the convex
counterparts of the nodes and edges of the manipulation graph
(Section~\ref{sec:method:graph}). The planning stage selects the keyframes
\eqref{eq:keyframes} with the minimum number of grasps
(Section~\ref{sec:method:sequencing}) and connects them with transit and
transfer trajectories (Section~\ref{sec:method:gcs}). The offline stage
depends only on the robot, object and environment, and can be reused across
queries with different $\qinit$ and $\qgoal$.

\begin{algorithm}[t]
\caption{Manipulation planner}
\label{alg:planner}
\begin{algorithmic}[1]
\Input robot, object, obstacles, manipulation constraints $\mathcal{M}$,
  $\qinit$, $\qgoal$, max.\ grasps $L$
\Output trajectory $\tau$
\State $\{P_j\}_{j=1}^{J} \gets \textsc{CliqueCoverIRIS}(\CSfree)$
  \Comment{Sec.~\ref{sec:method:decomp}}
\State $\{G_k\}_{k=1}^{K} \gets \textsc{GraspPolytopes}()$
  \Comment{Sec.~\ref{sec:method:grasp}, Alg.~\ref{alg:grasp}}
\State $\CPfree,\CGfree,\text{components} \gets \textsc{Intersect}(\{P_j\},\{G_k\},\CP)$
  \Comment{Sec.~\ref{sec:method:graph}}
\For{$\ell = 1,\dots,L$}
  \State $x_{0:2\ell+1} \gets \textsc{Sequence}(\ell,\CPfree,\CGfree,\mathcal{M})$
  \Comment{Sec.~\ref{sec:method:sequencing}}
  \If{a sequence is found} \textbf{break} \EndIf
\EndFor
\For{$i = 0,\dots,2\ell$}
  \State $\sigma \gets$ transit if $i$ even, else transfer
  \State $\tau_i \gets \textsc{GCS}(x_i, x_{i+1}; \text{regions of mode }\sigma, \mathcal{M}_\sigma)$
  \Comment{Sec.~\ref{sec:method:gcs}}
\EndFor
\State \Return $\tau = \tau_0\oplus\dots\oplus\tau_{2\ell}$
\end{algorithmic}
\end{algorithm}

\subsection{Convex decomposition of the collision-free space}
\label{sec:method:decomp}
We compute a collection of convex polytopes
$P_j=\{q\in\R^{n} : A_j q\le b_j\}$, $j=1,\dots,J$, whose union
$\Pfree=\bigcup_j P_j$ is an inner approximation of $\CSfree$. We use clique
covers of visibility graphs~\cite{werner2024approximating}: collision-free
configurations are sampled, a visibility graph is built between them, it is
covered by cliques, and each clique seeds the growth of one region, here with
IRIS-NP~\cite{petersen2023growing} as implemented in
Drake~\cite{tedrake2019drake}. The process repeats until a fraction $\alpha$ of
sampled collision-free configurations is covered. Regions grown this way are
collision-free with high probability; when certification is required, they can
be mapped to the tangent configuration space $t_i=\tan(\theta_i/2)$ of each
revolute joint and certified, or shrunk until certified, with
C-IRIS~\cite{dai2024certified}. \tocheck{the MInf pipeline maps polytope
vertices to TC-space, takes the convex hull, shrinks with C-IRIS and maps
vertices back; the back-mapped hull is not guaranteed to stay inside the
certified set, since the map is nonlinear. Either fix, state as approximate, or
drop certification from the 10-DOF results.} The decomposition is computed
once per scene, and is the only stage whose cost grows directly with the
volume of free space covered. \todo{coverage $\alpha$ and IRIS/clique-cover
parameters per task; the stored 10-DOF decomposition has 18 regions at

\subsection{Grasp and placement polytopes}
\label{sec:method:grasp}
The planner needs $\CP$ and $\CG$ as unions of polytopes,
$\CP=\bigcup_{k'} P^{\mathrm{pl}}_{k'}$ and $\CG\supseteq\bigcup_{k} G_k$.
In the bottle-cap tasks, the cap rotates about a fixed axis and is always
stably placed, so $\CP$ is the joint-limit box. Grasps are where the
two settings differ.

\paragraph{Low-DOF grippers}
When the gripper joints map directly to the grasp, $\CG$ is a box in
configuration space. For the 3- and 4-DOF tasks of
Section~\ref{sec:experiments}, a grasp is any configuration in which the
finger touches the cap, $z_{\mathrm{finger}}=-0.025$\,m, with the other joints
free within their limits. To give the set non-zero volume we use a 1\,mm band,
$z_{\mathrm{finger}}\in[-0.025,-0.024]$, so $\CG$ is a single box, i.e., the
convex hull of its $2^{n}$ corners.

\paragraph{High-DOF arms}
For a 7-DOF arm, a grasp constrains the pose of the hand frame $E$ relative to
the cap frame $C$, ${}^{C}X^{E}(q)=({}^{C}R^{E}(q),{}^{C}p^{E}(q))$, which is
a nonlinear function of the arm joints. For the cap task, the hand must be
centred above the cap, within a height band, with its approach axis
$z_E={}^{C}R^{E}(q)\,e_z$ pointing down along the cap axis:
\begin{align}
  h(q) &= \big[\,p_x,\; p_y,\; z_{E,x},\; z_{E,y}\,\big]^\top = 0, \label{eq:grasp_h}\\
  g(q) &= \big[\,p_z - z_{\max},\; z_{\min} - p_z,\; z_{E,z} + c\,\big]^\top \le 0, \label{eq:grasp_g}
\end{align}
with $p={}^{C}p^{E}(q)$, $[z_{\min},z_{\max}]=[0.105,0.110]$\,m and $c=0.9$,
together with the finger joints restricted to a 1\,mm band around the cap
width. The set $\{q: h(q)=0,\,g(q)\le0\}$ is a thin, curved manifold of
dimension $n-4$ in $\CS$, which cannot be written down as a polytope. We
approximate it with a union of thin polytopes around diverse grasp
configurations (Algorithm~\ref{alg:grasp}).

\begin{algorithm}[t]
\caption{Grasp polytopes for a high-DOF arm}
\label{alg:grasp}
\begin{algorithmic}[1]
\Input grasp constraints $h,g$ and Jacobians; joint limits; margins $\epsilon$, tolerance $\eta$
\Output grasp polytopes $\{G_k\}$
\State $\{q_k\} \gets \textsc{NLPSampling}(h,g)$ \Comment{diverse grasp seeds}
\For{each seed $q_k$}
  \State $J_k \gets [\,\partial h/\partial q;\; \partial g_{\mathcal{A}}/\partial q\,](q_k)$ \Comment{active rows}
  \State $N_k \gets \mathrm{null}(J_k)$, $\;V_k \gets N_k^{\perp}$ \Comment{orthonormal bases}
  \State $r_k \gets$ largest extents along $N_k$ s.t.\ the box validates
  \State $G_k \gets \{q : |N_k^\top(q-q_k)|\le r_k,\ |V_k^\top(q-q_k)|\le\epsilon\}$
  \State validate $G_k$: $\|h\|_\infty\le\eta$, $g\le\eta$ on corners and random points
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{Grasp seeds with NLP sampling}
Grasp polytopes are only useful if their seeds cover the different
components of the grasp manifold (e.g., different elbow postures). A local
inverse-kinematics solver returns one solution per initial guess and gives no
control over diversity. We instead use NLP sampling~\cite{toussaint2024nlp},
which targets the density $p(q)\propto e^{-f(q)}\,\mathbb{1}[g(q)\le0]\,\mathbb{1}[h(q)=0]$,
here with $f\equiv0$. Each restart (i) picks a seed among candidates far from
all previous samples, (ii) runs a damped Gauss--Newton descent on the slack
$\|s(q)\|^2$, $s=[\max(g,0);\,h]$, until a feasible point is found, and
(iii) explores the feasible set from it with an interior sampler, relaxing
$h=0$ to a tube $|h|\le\eta_{w}$ and projecting proposal directions onto the
tangent space,
$d\gets\big(I-J_h^\top(J_hJ_h^\top+\lambda I)^{-1}J_h\big)d$. A final
Gauss--Newton projection brings every sample within $\|h\|_\infty\le10^{-3}$.
Our implementation is a Python port of the reference implementation
of~\cite{toussaint2024nlp}, with nonlinear hit-and-run (NHR) and manifold RRT
(mRRT) as interior methods, and Drake's automatic differentiation for the
Jacobians. On the grasp problem, NHR mixed poorly in the wrist and cap
coordinates: its line search shares one step size across all coordinates, so
the tightest constraint throttles motion even along directions it does not
depend on. We diagnose this by comparing the spread of each coordinate within
a chain with its spread across restarts, and switch to mRRT, which takes
fixed-length tangent steps and corrects them afterwards.

\paragraph{Null-space polytopes}
Around each seed $q_k$, we linearise the active constraints, i.e., $h$ and the
rows of $g$ within $\delta$ of zero, and compute an orthonormal basis $N_k$ of
the null space of their Jacobian $J_k$ and a basis $V_k$ of its orthogonal
complement. Directions in $N_k$ preserve the grasp to first order, and some of
them preserve it exactly: rotating the wrist, whose axis passes through the
origin of $E$ along $z_E$, changes neither $p$ nor $z_E$; the cap joint and
the fingers do not enter \eqref{eq:grasp_h}--\eqref{eq:grasp_g} at all. These
directions are given their full joint range. The remaining null direction of
the 7-DOF arm, a self-motion that mixes all shoulder and elbow joints, is
given the largest extent for which the constraints remain within a tolerance
$\eta$, and the complement $V_k$ a small margin $\epsilon$. The resulting
$G_k$ is a box in rotated coordinates, i.e., an H-polytope with $2n$
half-spaces, intersected with the joint limits. We validate each $G_k$ by
checking the constraints at the corners of the non-trivial directions and at
random interior points.

Because the wrist and cap are exact invariant directions of the grasp
constraint, a transfer motion for the cap task (a coupled rotation of wrist
and cap, \eqref{eq:transfer}) stays within a single $G_k$; regrasps between
different grasp polytopes happen through transit motions.

\subsection{Mode sets and their connectivity}
\label{sec:method:graph}
We intersect the grasp and placement spaces with the decomposition,
\begin{equation}
  \CPfree = \bigcup_{j}\big(\CP\cap P_j\big),\qquad
  \CGfree = \bigcup_{j,k}\big(G_k\cap P_j\big),
  \label{eq:modefree}
\end{equation}
keeping only non-empty intersections (each checked with one linear program).
Both are unions of convex polytopes, possibly disconnected. We compute their
connected components from the intersection graph of their polytopes: two
polytopes are adjacent if they intersect. The components of $\CPfree$ and
$\CGfree$ play the role of the transit and transfer edges of the manipulation
graph: two configurations in the same component of $\CPfree$ can be connected
by a collision-free path in $\CP$, and similarly for $\CGfree$. Configurations
in $\CGfree\cap\CPfree$ are candidate grasps and releases, the nodes of the
manipulation graph~\cite{simeon2004manipulation}.

\subsection{Grasp--release sequencing}
\label{sec:method:sequencing}

\paragraph{Mixed-integer formulation}
For a fixed number of grasps $\ell$, we choose the keyframes
\eqref{eq:keyframes} by solving
\begin{equation}
\begin{aligned}
  \min_{x_0,\dots,x_{2\ell+1}}\ & \textstyle\sum_{i=0}^{2\ell}\|x_{i+1}-x_i\|_2^2\\
  \text{s.t.}\ \ & x_0=\qinit,\quad x_{2\ell+1}=\qgoal,\quad q_{\min}\le x_i\le q_{\max},\\
  & \mathcal{M}_{\mathrm{tr}}(x_{2i},x_{2i+1}),\ \ x_{2i}\sim_{\CPfree}x_{2i+1},\ \ i\le\ell,\\
  & \mathcal{M}_{\mathrm{tf}}(x_{2i+1},x_{2i+2}),\ \ x_{2i+1}\sim_{\CGfree}x_{2i+2},\ \ i<\ell,
\end{aligned}
\label{eq:miqp}
\end{equation}
where $x\sim_{\mathcal{U}}x'$ requires both points to lie in the same
connected component of the union of polytopes $\mathcal{U}$. For
$\mathcal{U}=\bigcup_m Q_m$, $Q_m=\{x:A_mx\le b_m\}$, with components
$\mathcal{U}_1,\dots,\mathcal{U}_C$, we encode membership of each point $x$
with binaries $\delta_m$ (point in $Q_m$) and $\lambda_c$ (point in
$\mathcal{U}_c$):
\begin{equation}
\begin{gathered}
  A_m x \le b_m + M(1-\delta_m),\quad \textstyle\sum_m \delta_m \ge 1,\\
  \textstyle\sum_c \lambda_c = 1,\quad \delta_m \le \lambda_{c(m)},
\end{gathered}
\label{eq:bigm}
\end{equation}
and require the component indicators $\lambda$ of the two points of a pair to
be equal. Since every grasp and release keyframe belongs to a transit pair and
a transfer pair, it lies in $\CGfree\cap\CPfree$. The objective, a sum of
squared keyframe distances, is a convex surrogate for the length of the
final trajectory. Problem \eqref{eq:miqp} is a mixed-integer convex quadratic
program, which we solve with MOSEK~\cite{mosek} for $\ell=1,2,\dots$ until the
first feasible $\ell$, which gives the minimum number of grasps (relative to
the decomposition, see Proposition~\ref{prop:guarantees}).

On the 10-DOF task, the MIQP fails to return feasible sequences even where
they exist. We attribute this to numerics: the grasp polytopes have widths of
order $\epsilon=10^{-3}$, while the big-M constant must dominate the
constraint violation over the whole joint range, and the problem grows with
$(2\ell+2)$ copies of all region binaries.
\tocheck{confirm cause; possible fixes: per-row tight $M$ from joint limits,
convex-hull (perspective) formulation, rescaling the thin directions.}
This motivates the graph-search alternative below.

\subsection{Transit and transfer trajectories with GCS}
\label{sec:method:gcs}
Given the keyframes, each pair $(x_i,x_{i+1})$ is connected by a shortest
path in the graph of convex sets formed by the polytopes of the corresponding
mode, $\CPfree$ for transit and $\CGfree$ for transfer, using Drake's
\texttt{GcsTrajectoryOptimization}~\cite{marcucci2023motion,tedrake2019drake}.
Each trajectory segment is a B\'ezier curve whose control points lie in one
polytope, so the whole curve lies in that polytope and in $\Pfree$. When the
manipulation constraints are linear, as in \eqref{eq:transit}--\eqref{eq:transfer},
we impose them on all control points, and since a B\'ezier curve is a convex
combination of its control points, they then hold along the entire segment.

\subsection{Guarantees}
\label{sec:method:guarantees}
\begin{proposition}[informal]
\label{prop:guarantees}
Let $\Pfree\subseteq\CSfree$. (i) Any trajectory returned by
Algorithm~\ref{alg:planner} is collision-free, respects joint limits and
satisfies the linear manipulation constraints along every segment.
(ii) Problem~\eqref{eq:miqp} is a relaxation of the planning problem
restricted to the represented sets: every valid manipulation path with $\ell$
grasps whose transit segments lie in $\CPfree$ and transfer segments in
$\CGfree$ yields a feasible keyframe sequence for~\eqref{eq:miqp}.
Hence, if \eqref{eq:miqp} is infeasible for all $\ell\le L$, no such path with
at most $L$ grasps exists within the decomposition, and if the GCS stage
succeeds for the first feasible $\ell$, the plan uses the minimum number of
grasps relative to the decomposition.
\end{proposition}
\noindent\emph{Sketch.} (i) follows from the containment of each B\'ezier
segment in a polytope of $\Pfree$ and the convex-combination argument above.
(ii) The keyframes of a valid path satisfy the constraints of
\eqref{eq:miqp}, since consecutive keyframes of a transit (transfer) segment are
connected within $\CPfree$ ($\CGfree$) and therefore lie in the same component.
The converse does not hold: two configurations in the same component of
$\CPfree$ need not be connected by a path that keeps $\qcap$ fixed, so the GCS
stage can fail on a feasible sequence. \tocheck{state precisely; relates to
the 4-DOF, $\alpha=0.90$ case in Table~\ref{tab:lowdof}.}
Optimality of the path length is only partial: each segment is optimal
relative to the decomposition given its endpoints when the convex relaxation
of GCS is tight~\cite{marcucci2024shortest,marcucci2023motion}, but the
keyframes are chosen with a surrogate cost.
\hfill$\square$
